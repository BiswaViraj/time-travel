# or
pnpm add @biswaviraj/time-travel`;

export const quickExample = `import { timeTravel } from "@biswaviraj/time-travel";

const tt = timeTravel<number>(0);

tt.add(1);
tt.add(2);
tt.add(3);

tt.get();    // 3
tt.undo();   // returns 2
tt.undo();   // returns 1
tt.redo();   // returns 2

tt.canUndo;  // true
tt.canRedo;  // true`;
